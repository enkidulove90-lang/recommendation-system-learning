# 2916. 子数组不同元素数目的平方和 II

话题标签: binary-indexed-tree, segment-tree, array, hash-table

## 思路

题目要算所有子数组「不同元素个数」的平方和。直接枚举所有子数组是 O(n²) 起步，n 到 1e5 时不可行，需要换个角度。

固定右端点 r，考虑所有以 r 结尾的子数组 nums[l..r]（l 从 0 到 r）。记 d(l) 为 nums[l..r] 的不同计数。当我们把右端点从 r-1 推进到 r 时，新加入的元素 x = nums[r] 会怎样影响每个 d(l)？设 x 上一次出现的下标是 p = last[x]，那么对于 l ≤ p，区间 nums[l..r] 里早就含过 x 了，不同计数不变；对于 l 在 [p+1, r]，这是这些区间里第一次出现 x，不同计数都要加 1。也就是说，推进右端点等价于对数组 d 的一段后缀区间做「整体加 1」。

这样我们就把问题转化成了：维护一个长度为 n 的数组 d（初值 0），每步对区间 [p+1, r] 做区间加 1，然后查询整个数组的平方和，累加到答案里。区间加、区间平方和查询正好可以用带懒标记的线段树维护。

关键在于平方和如何在区间加时快速更新。设某区间元素和为 S、平方和为 Q，整体加 v 后：S' = S + v·len，Q' = Σ(aᵢ+v)² = Σaᵢ² + 2v·Σaᵢ + v²·len = Q + 2v·S + v²·len。所以只要同时维护区间和与区间平方和，加标记就能 O(1) 更新。

## 解题过程

1. 建一棵线段树，每个节点维护 `sum_tree`（区间元素和）、`sq_tree`（区间元素平方和）、`lazy`（待下传的加标记），下标范围 [0, n-1]，初值全为 0。
2. 用 `last[x]` 记录值 x 上一次出现的下标，初始为 -1。
3. 从左到右枚举右端点 r 和值 x：
   - 计算 p = last[x]，对区间 [p+1, r] 整体加 1。含义是：这些左端点对应的子数组是第一次包含 x。
   - 把根节点的 `sq_tree[1]`（即当前所有以 r 结尾子数组不同计数的平方和）累加到答案。
   - 更新 `last[x] = r`。
4. 返回答案对 1e9+7 取模的结果。

以 nums = [1,2,1] 为例：
- r=0, x=1, p=-1：区间 [0,0] 加 1，d=[1]，平方和 1，ans=1。
- r=1, x=2, p=-1：区间 [0,1] 加 1，d=[2,1]，平方和 5，ans=6。
- r=2, x=1, p=0：区间 [1,2] 加 1，d=[2,2,1]，平方和 9，ans=15。与示例一致。

线段树的 `apply` 负责在整段被覆盖时更新和、平方和与懒标记；`push` 在向下递归前把标记下传给左右儿子；`update` 递归处理部分覆盖的区间。

## 复杂度

时间复杂度：O(n log n)，每个右端点做一次区间加和一次根节点读取，区间加在线段树上是 O(log n)。
空间复杂度：O(n)，线段树三个数组各开 4n 大小，last 数组为固定大小。

## Code

```python3
import sys

sys.setrecursionlimit(1 << 20)

class Solution:
    def sumCounts(self, nums: list[int]) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        
        # 线段树数组：sum_tree 维护区间和，sq_tree 维护区间平方和，lazy 为加标记
        sum_tree = [0] * (4 * n + 5)
        sq_tree = [0] * (4 * n + 5)
        lazy = [0] * (4 * n + 5)
        
        def apply(node: int, l: int, r: int, v: int) -> None:
            """区间 [l, r] 所有元素加上 v"""
            length = r - l + 1
            # (sum + v)^2 = sum^2 + 2*v*sum + v^2 * length
            sq_tree[node] = (sq_tree[node] + 2 * v * sum_tree[node] + v * v * length) % MOD
            sum_tree[node] = (sum_tree[node] + v * length) % MOD
            lazy[node] = (lazy[node] + v) % MOD
        
        def push(node: int, l: int, r: int) -> None:
            """下传 lazy 标记"""
            if lazy[node]:
                mid = (l + r) >> 1
                v = lazy[node]
                apply(node << 1, l, mid, v)
                apply(node << 1 | 1, mid + 1, r, v)
                lazy[node] = 0
        
        def update(node: int, l: int, r: int, ql: int, qr: int, v: int) -> None:
            """区间更新：将 [ql, qr] 内所有元素加 v"""
            if ql <= l and r <= qr:
                apply(node, l, r, v)
                return
            push(node, l, r)
            mid = (l + r) >> 1
            if ql <= mid:
                update(node << 1, l, mid, ql, qr, v)
            if qr > mid:
                update(node << 1 | 1, mid + 1, r, ql, qr, v)
            sum_tree[node] = (sum_tree[node << 1] + sum_tree[node << 1 | 1]) % MOD
            sq_tree[node] = (sq_tree[node << 1] + sq_tree[node << 1 | 1]) % MOD
        
        last = [-1] * 100001   # 记录每个值上一次出现的下标
        ans = 0
        
        for r, x in enumerate(nums):
            l = last[x] + 1    # 对于左端点 l ∈ [last[x]+1, r]，x 是新出现的，不同计数 +1
            update(1, 0, n - 1, l, r, 1)
            ans = (ans + sq_tree[1]) % MOD   # 根节点的 sq 即所有以 r 结尾子数组的平方和
            last[x] = r
        
        return ans
```

### C++

```cpp
#include <vector>
#include <functional>

class Solution {
public:
    int sumCounts(std::vector<int>& nums) {
        const long long MOD = 1000000007LL;
        int n = nums.size();

        std::vector<long long> sum_tree(4 * n + 5, 0);
        std::vector<long long> sq_tree(4 * n + 5, 0);
        std::vector<long long> lazy(4 * n + 5, 0);

        std::function<void(int, int, int, long long)> apply = [&](int node, int l, int r, long long v) {
            long long length = r - l + 1;
            sq_tree[node] = (sq_tree[node] + 2 * v * sum_tree[node] + v * v * length) % MOD;
            sum_tree[node] = (sum_tree[node] + v * length) % MOD;
            lazy[node] = (lazy[node] + v) % MOD;
        };

        std::function<void(int, int, int)> push = [&](int node, int l, int r) {
            if (lazy[node]) {
                int mid = (l + r) >> 1;
                long long v = lazy[node];
                apply(node << 1, l, mid, v);
                apply(node << 1 | 1, mid + 1, r, v);
                lazy[node] = 0;
            }
        };

        std::function<void(int, int, int, int, int, long long)> update = [&](int node, int l, int r, int ql, int qr, long long v) {
            if (ql <= l && r <= qr) {
                apply(node, l, r, v);
                return;
            }
            push(node, l, r);
            int mid = (l + r) >> 1;
            if (ql <= mid) {
                update(node << 1, l, mid, ql, qr, v);
            }
            if (qr > mid) {
                update(node << 1 | 1, mid + 1, r, ql, qr, v);
            }
            sum_tree[node] = (sum_tree[node << 1] + sum_tree[node << 1 | 1]) % MOD;
            sq_tree[node] = (sq_tree[node << 1] + sq_tree[node << 1 | 1]) % MOD;
        };

        std::vector<int> last(100001, -1);
        long long ans = 0;

        for (int r = 0; r < n; r++) {
            int x = nums[r];
            int l = last[x] + 1;
            update(1, 0, n - 1, l, r, 1);
            ans = (ans + sq_tree[1]) % MOD;
            last[x] = r;
        }

        return (int)ans;
    }
};
```
